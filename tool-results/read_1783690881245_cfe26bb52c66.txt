   276→   276→function GalleryCard({
   277→   277→  image,
   278→   278→  index,
   279→   279→  isInView,
   280→   280→  onClick,
   281→   281→}: {
   282→   282→  image: GalleryItem;
   283→   283→  index: number;
   284→   284→  isInView: boolean;
   285→   285→  onClick: () => void;
   286→   286→}) {
   287→   287→  const [tiltStyle, setTiltStyle] = useState<React.CSSProperties>({});
   288→   288→  const [isHovered, setIsHovered] = useState(false);
   289→   289→
   290→   290→  const handleMouseMove = (e: MouseEvent<HTMLDivElement>) => {
   291→   291→    const rect = e.currentTarget.getBoundingClientRect();
   292→   292→    const x = e.clientX - rect.left;
   293→   293→    const y = e.clientY - rect.top;
   294→   294→    const centerX = rect.width / 2;
   295→   295→    const centerY = rect.height / 2;
   296→   296→    const rotateY = ((x - centerX) / centerX) * 8;
   297→   297→    const rotateX = ((centerY - y) / centerY) * 8;
   298→   298→
   299→   299→    setTiltStyle({
   300→   300→      transform: `perspective(800px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(1.05, 1.05, 1.05)`,
   301→   301→      transition: "transform 0.2s cubic-bezier(0.03, 0.98, 0.52, 0.99)",
   302→   302→      transformStyle: "preserve-3d",
   303→   303→    });
   304→   304→  };
   305→   305→
   306→   306→  const handleMouseLeave = () => {
   307→   307→    setTiltStyle({
   308→   308→      transform:
   309→   309→        "perspective(800px) rotateX(0deg) rotateY(0deg) scale3d(1, 1, 1)",
   310→   310→      transition: "transform 0.5s cubic-bezier(0.03, 0.98, 0.52, 0.99)",
   311→   311→    });
   312→   312→    setIsHovered(false);
   313→   313→  };
   314→   314→
   315→   315→  return (
   316→   316→    <motion.div
   317→   317→      key={`gallery-${index}`}
   318→   318→      className="perspective-1000"
   319→   319→      initial={{ opacity: 0, scale: 0.9 }}
   320→   320→      animate={isInView ? { opacity: 1, scale: 1 } : {}}
   321→   321→      transition={{ duration: 0.4, delay: index * 0.06 }}
   322→   322→    >
   323→   323→      <div
   324→   324→        className="group relative rounded-xl overflow-hidden shadow-md cursor-pointer aspect-square shine-3d"
   325→   325→        style={tiltStyle}
   326→   326→        onMouseMove={handleMouseMove}
   327→   327→        onMouseEnter={() => setIsHovered(true)}
   328→   328→        onMouseLeave={handleMouseLeave}
   329→   329→        onClick={onClick}
   330→   330→      >
   331→   331→        <Image
   332→   332→          src={image.src}
   333→   333→          alt={image.alt}
   334→   334→          fill
   335→   335→          className="object-cover group-hover:scale-110 transition-transform duration-700"
   336→   336→        />
   337→   337→        <div className="absolute inset-0 bg-gradient-to-t from-teal-900/80 via-teal-900/20 to-transparent opacity-0 group-hover:opacity-100 transition-opacity duration-300" />
   338→   338→
   339→   339→        {/* 3D floating icon badge */}
   340→   340→        <div
   341→   341→          className="absolute top-2 right-2 w-7 h-7 bg-white/90 backdrop-blur-sm rounded-lg flex items-center justify-center opacity-0 group-hover:opacity-100 transition-all duration-300 shadow-md"
   342→   342→          style={{
   343→   343→            transformStyle: "preserve-3d",
   344→   344→            transform: isHovered ? "translateZ(40px)" : "translateZ(0px)",
   345→   345→            transition: "transform 0.3s ease, opacity 0.3s ease",
   346→   346→          }}
   347→   347→        >
   348→   348→          <image.icon className="w-3.5 h-3.5 text-teal-600" />
   349→   349→        </div>
   350→   350→
   351→   351→        {/* Info overlay with 3D depth */}
   352→   352→        <div
   353→   353→          className="absolute bottom-0 left-0 right-0 p-2.5 translate-y-full group-hover:translate-y-0 transition-transform duration-300"
   354→   354→          style={{ transformStyle: "preserve-3d" }}
   355→   355→        >
   356→   356→          <p
   357→   357→            className="text-white font-semibold text-xs leading-tight"
   358→   358→            style={{
   359→   359→              transform: isHovered ? "translateZ(30px)" : "translateZ(0px)",
   360→   360→              transition: "transform 0.3s ease 0.1s",
   361→   361→            }}
   362→   362→          >
   363→   363→            {image.name || image.alt}
   364→   364→          </p>
   365→   365→          <p
   366→   366→            className="text-teal-200 text-[10px] mt-0.5"
   367→   367→            style={{
   368→   368→              transform: isHovered ? "translateZ(20px)" : "translateZ(0px)",
   369→   369→              transition: "transform 0.3s ease 0.15s",
   370→   370→            }}
   371→   371→          >
   372→   372→            {image.specialty || image.info}
   373→   373→          </p>
   374→   374→        </div>
   375→   375→
   376→   376→        {/* 3D border glow on hover */}
   377→   377→        <div className="absolute inset-0 rounded-xl border-2 border-teal-400/0 group-hover:border-teal-400/30 transition-all duration-300 pointer-events-none" />
   378→   378→      </div>
   379→   379→    </motion.div>
   380→   380→  );
